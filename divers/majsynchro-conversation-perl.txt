my $logTempfile = "$G_SYDT/import_temp.log" ;

foreach my $file ( glob("$G_SYDD/*") ) {
    # Ignorer si ce n'est pas un fichier valide ou si c'est un répertoire
    next unless -f $file; 
    
    my $filename = basename($file);
    
    # Sécurisation de l'extraction du type, de la partition et de la version
    # Format attendu : YYYY-MM-DD_HH-MM-SS_TYPE_PARTITION@VERSION.hkp
    # $1 = Date/Heure, $2 = Type, $3 = Partition, $4 = Version (sans le .hkp)
    if ( $filename =~ /^(.{19})_(.*?)_(.*?)@([^.]+)/ ) {
        my $type          = $2;
        my $Partition     = $3;
        my $symbolversion = $4; # Servira aussi pour l'ibis_model
        
        # --- PADSTACK ---
        if ( $type eq "padstack" ) {
            system("export DISPLAY=$MGCDISPLAY ; source $MGCENV ; HKP2PadstackDB -m -l $logTempfile -i $file -o $G_MDULMC/Layout/PadstackDB.psk -c $G_MDFULMC") ;
            if ($? != 0) { 
                WLOG(SYN001, $logfile);
            } else {
                unlink($file);
            }
        }
        
        # --- CELL ---
        elsif ( $type eq "cell" ) {      
            system("export DISPLAY=$MGCDISPLAY ; source $MGCENV ; HKP2CellDB -m -l $logTempfile -i $file -o $G_MDULMC/CellDBLibs/$Partition.cel -c $G_MDFULMC") ;
            if ($? != 0) { 
                WLOG(SYN002, $logfile);
            } else {
                unlink($file);
            }
        }
        
        # --- SYMBOL ---
        elsif ( $type eq "symbol" ) { 
            system("cp $file $G_MDULMC/SymbolLibs/$Partition/sym/$symbolversion");
            if ($? != 0) {  
                WLOG(SYN003, $logfile);
            } else {
                system("export DISPLAY=$MGCDISPLAY ; source $MGCENV ; epdrun dxlib2lmc -centrallib=$G_MDFULMC -partition=$Partition -Symbol=$symbolversion -logfile=$logTempfile") ;
                if ($? != 0) { 
                    WLOG(SYN003, $logfile);
                } else {
                    unlink($file);
                }
            }
        }
        
        # --- PART ---
        elsif ( $type eq "part" ) {     
            system("export DISPLAY=$MGCDISPLAY ; source $MGCENV ; HKP2PartsDB -r -l $logTempfile -i $file -o $G_MDULMC/PartsDBLibs/$Partition.pdb -c $G_MDFULMC") ;
            if ($? != 0) { 
                WLOG(SYN004, $logfile);
            } else {
                unlink($file);
            }
        }
        
        # --- IBIS ---
        elsif ( $type eq "ibis" ) {
            system("cp $file $G_MDULMC/IBISModels/$Partition/$symbolversion");
            if ($? != 0) { 
                WLOG(SYN004, $logfile);
            } else {
                unlink($file);
            }
        }
    }
    # Gestion optionnelle si le fichier n'a pas de "@" (ex: padstack qui n'a parfois pas de partition)
    elsif ( $filename =~ /^(.{19})_padstack_(.*)/ ) {
        system("export DISPLAY=$MGCDISPLAY ; source $MGCENV ; HKP2PadstackDB -m -l $logTempfile -i $file -o $G_MDULMC/Layout/PadstackDB.psk -c $G_MDFULMC") ;
        if ($? != 0) { WLOG(SYN001, $logfile); } else { unlink($file); }
    }
}
