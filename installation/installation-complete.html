1. node_exporter

1.1 Script d'installation

install_node_exporter.sh
#!/bin/bash
set -euo pipefail

fail() { echo "ERREUR: $1" >&2; exit 1; }

ARCHIVE="$1"
[ -z "$ARCHIVE" ] && fail "Usage: $0 <archive_node_exporter.tar.gz>"
[ -f "$ARCHIVE" ] || fail "Archive introuvable: $ARCHIVE"

INSTALL_DIR="/usr/share/node_exporter"

id node_exporter >/dev/null 2>&1 || sudo useradd --no-create-home --shell /usr/sbin/nologin node_exporter

EXTRACT_DIR=$(tar -tzf "$ARCHIVE" | head -1 | cut -f1 -d"/")
[ -z "$EXTRACT_DIR" ] && fail "Impossible de determiner le dossier d'extraction"

TMPDIR=$(mktemp -d)
tar -xzf "$ARCHIVE" -C "$TMPDIR" || fail "Extraction echouee"

sudo mkdir -p "$INSTALL_DIR"
sudo cp "$TMPDIR/$EXTRACT_DIR/node_exporter" "$INSTALL_DIR/node_exporter"
sudo chown node_exporter:node_exporter "$INSTALL_DIR/node_exporter"
sudo chmod 755 "$INSTALL_DIR/node_exporter"
sudo restorecon -Rv "$INSTALL_DIR/" || true

sudo tee /etc/systemd/system/node_exporter.service > /dev/null << 'EOF'
[Unit]
Description=Node Exporter
After=network.target

[Service]
User=node_exporter
Group=node_exporter
Type=simple
ExecStart=/usr/share/node_exporter/node_exporter
Restart=on-failure

[Install]
WantedBy=multi-user.target
EOF

sudo systemctl daemon-reload
sudo systemctl enable --now node_exporter

sudo firewall-cmd --permanent --add-port=9100/tcp 2>/dev/null || true
sudo firewall-cmd --reload 2>/dev/null || true

rm -rf "$TMPDIR"
echo "OK - node_exporter installe"

2. process-exporter

2.1 Script d'installation complet

install_process_exporter.sh
#!/bin/bash
set -euo pipefail

fail() { echo "ERREUR: $1" >&2; exit 1; }

[ "$(id -u)" -eq 0 ] || fail "Ce script doit etre lance avec sudo/root."

ARCHIVE="${1:-}"
[ -z "$ARCHIVE" ] && fail "Usage: $0 <archive_process-exporter.tar.gz>"
[ -f "$ARCHIVE" ] || fail "Archive introuvable: $ARCHIVE"

id node_exporter >/dev/null 2>&1 || fail "Le compte systeme 'node_exporter' n'existe pas - installer node_exporter d'abord."

INSTALL_DIR="/usr/share/process_exporter"
CONFIG_DIR="/etc/process_exporter"

TMPDIR=$(mktemp -d)

if command -v tar >/dev/null 2>&1; then
  tar -xzf "$ARCHIVE" -C "$TMPDIR" || fail "Extraction echouee (tar)"
elif command -v python3 >/dev/null 2>&1; then
  python3 -c "import tarfile,sys; tarfile.open(sys.argv[1]).extractall(sys.argv[2])" "$ARCHIVE" "$TMPDIR" \
    || fail "Extraction echouee (python3 tarfile)"
else
  fail "Ni 'tar' ni 'python3' disponibles - impossible d'extraire l'archive."
fi

BINARY_PATH=$(find "$TMPDIR" -type f -name "process-exporter" | head -1)
[ -z "$BINARY_PATH" ] && fail "Binaire 'process-exporter' introuvable dans l'archive"

mkdir -p "$INSTALL_DIR" "$CONFIG_DIR"
cp "$BINARY_PATH" "$INSTALL_DIR/process-exporter"
chown -R node_exporter:node_exporter "$INSTALL_DIR"
chmod 755 "$INSTALL_DIR" "$INSTALL_DIR/process-exporter" "$CONFIG_DIR"
restorecon -Rv "$INSTALL_DIR" 2>/dev/null || true
rm -rf "$TMPDIR"

cat > "$CONFIG_DIR/config.yml" << 'EOF'
process_names:
  - name: "{{.ExeBase}}:{{.Username}}"
    cmdline:
    - '.+'
EOF
chmod 644 "$CONFIG_DIR/config.yml"

cat > /etc/systemd/system/process_exporter.service << 'EOF'
[Unit]
Description=Process Exporter
After=network.target

[Service]
User=node_exporter
Group=node_exporter
Type=simple
ExecStart=/usr/share/process_exporter/process-exporter \
  --config.path /etc/process_exporter/config.yml \
  --web.listen-address=:9256
Restart=on-failure

[Install]
WantedBy=multi-user.target
EOF

if command -v firewall-cmd >/dev/null 2>&1 && systemctl is-active --quiet firewalld; then
  firewall-cmd --permanent --add-port=9256/tcp
  firewall-cmd --reload
fi

mkdir -p /var/lib/node_exporter/textfile_collector
chmod 755 /var/lib/node_exporter
chown node_exporter:node_exporter /var/lib/node_exporter/textfile_collector
chmod 755 /var/lib/node_exporter/textfile_collector

systemctl daemon-reload
systemctl enable --now process_exporter

echo "--- Verification ---"
sleep 3
curl -s localhost:9256/metrics | grep -c namedprocess_namegroup_cpu_seconds_total || true

2.2 Config process-exporter

/etc/process_exporter/config.yml
process_names:
  - name: "{{.ExeBase}}:{{.Username}}"
    cmdline:
    - '.+'
Utilise .ExeBase, pas .Comm — le noyau Linux tronque .Comm à 15 caractères (ex: "process_exporte" au lieu de "process_exporter"), ce qui aurait donné des noms de process faux et incohérents.

2.3 Résolution des utilisateurs AD

generate_uid_map.sh
#!/bin/bash
# Lookup CIBLE par UID (pas d'enumeration globale - SSSD n'enumere pas l'AD par defaut)
set -euo pipefail

OUTPUT_DIR="/var/lib/node_exporter/textfile_collector"
OUTPUT_FILE="$OUTPUT_DIR/uid_username_map.prom"
TMP_FILE="${OUTPUT_FILE}.tmp"

mkdir -p "$OUTPUT_DIR"

UIDS=$(curl -s localhost:9256/metrics 2>/dev/null \
  | grep -oP 'groupname="[^"]*:\K[0-9]+(?=")' \
  | sort -u)

{
  echo "# HELP node_uid_username Correspondance UID vers nom d'utilisateur (lookup cible via getent/SSSD)"
  echo "# TYPE node_uid_username gauge"
  for uid in $UIDS; do
    entry=$(getent passwd "$uid" 2>/dev/null || true)
    if [ -n "$entry" ]; then
      user=$(echo "$entry" | cut -d: -f1)
      user_escaped=$(printf '%s' "$user" | sed 's/"/\\"/g')
      echo "node_uid_username{uid=\"$uid\",username=\"$user_escaped\"} 1"
    fi
  done
} > "$TMP_FILE"

mv -f "$TMP_FILE" "$OUTPUT_FILE"
chmod 644 "$OUTPUT_FILE"
Pourquoi un lookup ciblé par UID et pas un getent passwd global : SSSD ne fait pas d'énumération complète de l'annuaire AD par défaut (trop coûteux) — un getent passwd seul ne renverrait que les comptes locaux/déjà en cache. Un lookup individuel par UID déclenche toujours une vraie requête SSSD et fonctionne.

Piège du mv interactif : sur certains comptes, mv est aliasé en mv -i et demande confirmation — invisible en usage manuel, mais bloque le script indéfiniment sous cron (personne pour répondre). D'où le mv -f explicite.
Cron (sous l'utilisateur node_exporter, jamais root) : 
*/10 * * * * node_exporter /usr/local/bin/generate_uid_map.sh

3. BogoMips

generate_bogomips.sh
#!/bin/bash
set -euo pipefail

OUTPUT_DIR="/var/lib/node_exporter/textfile_collector"
OUTPUT_FILE="$OUTPUT_DIR/cpu_bogomips.prom"
TMP_FILE="${OUTPUT_FILE}.tmp"

mkdir -p "$OUTPUT_DIR"
chmod 755 "$OUTPUT_DIR" 2>/dev/null || true

BOGOMIPS_VALUES=$(grep -i '^bogomips' /proc/cpuinfo | awk -F: '{print $2}' | tr -d ' ')

if [ -z "$BOGOMIPS_VALUES" ]; then
  echo "Aucune valeur bogomips trouvee dans /proc/cpuinfo - abandon."
  exit 1
fi

TOTAL=$(echo "$BOGOMIPS_VALUES" | awk '{sum+=$1} END {printf "%.2f", sum}')
AVG=$(echo "$BOGOMIPS_VALUES" | awk '{sum+=$1; n++} END {printf "%.2f", sum/n}')

{
  echo "# HELP node_cpu_bogomips_total Somme des BogoMips de tous les coeurs logiques"
  echo "# TYPE node_cpu_bogomips_total gauge"
  echo "node_cpu_bogomips_total $TOTAL"
  echo "# HELP node_cpu_bogomips_avg BogoMips moyen par coeur logique"
  echo "# TYPE node_cpu_bogomips_avg gauge"
  echo "node_cpu_bogomips_avg $AVG"
} > "$TMP_FILE"

mv -f "$TMP_FILE" "$OUTPUT_FILE"
chmod 644 "$OUTPUT_FILE"
Cron : 
*/15 * * * * node_exporter /usr/local/bin/generate_bogomips.sh
Piège rencontré : si node_exporter tourne depuis longtemps sans redémarrage, le flag --collector.textfile.directory peut être présent dans le fichier de service sans être actif dans le process réel en mémoire (un changement de fichier service ne s'applique qu'au redémarrage). Toujours forcer un systemctl restart node_exporter après ajout du flag, même s'il semblait déjà présent.

5.1 Explication des 4 jobs

JobPortRôle
prometheus9090Prometheus se scrape lui-même (santé de l'outil)
node_exporter9100Métriques système (CPU/RAM/disque/réseau) de chaque machine
flexlm9319 (localhost)Licences FlexLM — un seul exporter centralisé sur saxfrc0154, pas par machine
process_exporter9256Détail CPU/RAM par processus et par utilisateur, sur chaque machine

5.2 Pourquoi le relabel_configs sur node_exporter et process_exporter

Ajoute un label host (nom de machine sans port) identique sur les deux jobs malgré leurs ports différents (9100 vs 9256) — extrait directement de __address__ au moment de la découverte de la cible, avant même le scrape. Permet à Grafana d'utiliser une seule variable $server simple pour filtrer les deux types de métriques, sans jonglage de port ni variable dépendante fragile (source de plusieurs bugs de redirection résolus grâce à ça).

5.3 Pourquoi le metric_relabel_configs sur process_exporter uniquement

Ne garde que les métriques namedprocess_namegroup_* réellement utilisées par les dashboards (CPU, RAM, nb instances, date de démarrage) + les métriques de santé de l'exporter lui-même — élimine tout le reste (write/read bytes, threads, métriques internes Go) pour réduire drastiquement le volume de séries stockées, sans rien changer côté node_exporter (moins de cardinalité par nature, pas besoin de filtrer).
